set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${runtime_manifest}")
file(READ "${runtime_manifest}" runtime_manifest_json)
string(JSON runtime_factory_count LENGTH "${runtime_manifest_json}" plugins)
set(TRAVERSE_REQUIRED_FACTORIES "")
math(EXPR runtime_factory_last "${runtime_factory_count} - 1")
foreach(index RANGE 0 ${runtime_factory_last})
  string(JSON factory GET "${runtime_manifest_json}" plugins ${index})
  if(NOT factory MATCHES "^[A-Za-z0-9_-]+$")
    message(FATAL_ERROR "Invalid runtime factory name")
  endif()
  string(APPEND TRAVERSE_REQUIRED_FACTORIES "\"${factory}\", ")
endforeach()
