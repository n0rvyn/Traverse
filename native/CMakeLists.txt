cmake_minimum_required(VERSION 3.20)
project(traverse VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-webrtc-1.0 gstreamer-sdp-1.0)
pkg_check_modules(SRTP REQUIRED IMPORTED_TARGET libsrtp2)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(X11 REQUIRED)
find_package(httplib CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(traverse-bind-helper bind_helper.c)
target_compile_options(traverse-bind-helper PRIVATE -Wall -Wextra -Wpedantic -Werror
  -fstack-protector-strong -fPIE)
target_link_options(traverse-bind-helper PRIVATE -pie -Wl,-z,relro,-z,now)
add_library(traverse-settings endpoint_settings.cpp)
target_include_directories(traverse-settings PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(traverse-settings PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(traverse-settings PUBLIC nlohmann_json::nlohmann_json)
add_library(traverse-tls-routing tls_router.cpp tls_certificate.cpp)
target_compile_options(traverse-tls-routing PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(traverse-tls-routing PUBLIC OpenSSL::SSL OpenSSL::Crypto Threads::Threads
  PRIVATE nlohmann_json::nlohmann_json)
# Shared public TLS listener; standalone supervisor integration selects it explicitly.
add_executable(traverse-tls-frontend tls_frontend.cpp)
target_link_libraries(traverse-tls-frontend PRIVATE traverse-tls-routing)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  pkg_check_modules(GST_CONVERSION_TEST REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-video-1.0)
  add_executable(conversion-quality-test ../tests/conversion_quality.cpp)
  target_include_directories(conversion-quality-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(conversion-quality-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(conversion-quality-test PRIVATE PkgConfig::GST_CONVERSION_TEST)
  add_test(NAME conversion-quality COMMAND conversion-quality-test)
  set_tests_properties(conversion-quality PROPERTIES TIMEOUT 60)
  add_executable(endpoint-settings-test ../tests/endpoint_settings_test.cpp)
  target_link_libraries(endpoint-settings-test PRIVATE traverse-settings)
  add_test(NAME endpoint-settings COMMAND endpoint-settings-test)
  add_test(NAME tls-router-sockets COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/../tests/tls_router_test.py $<TARGET_FILE:traverse-tls-frontend>
    $<TARGET_FILE:traverse-service>)
endif()
add_library(traverse-common INTERFACE)
target_compile_definitions(traverse-common INTERFACE GST_USE_UNSTABLE_API)
target_compile_options(traverse-common INTERFACE -Wall -Wextra -Wpedantic)
target_link_libraries(traverse-common INTERFACE httplib::httplib nlohmann_json::nlohmann_json OpenSSL::SSL OpenSSL::Crypto Threads::Threads PkgConfig::GST)
add_executable(traverse-adapter adapter.cpp input.cpp)
target_link_libraries(traverse-adapter PRIVATE traverse-common X11::X11 X11::Xtst)
add_executable(traverse-service service.cpp)
target_link_libraries(traverse-service PRIVATE traverse-common traverse-settings)
if(BUILD_TESTING)
  add_test(NAME loopback-service-boundaries COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/../tests/loopback_service_test.py $<TARGET_FILE:traverse-service>
    ${CMAKE_CURRENT_SOURCE_DIR}/../web)
endif()
add_executable(traverse-runtime-check runtime_check.cpp)
target_link_libraries(traverse-runtime-check PRIVATE nlohmann_json::nlohmann_json PkgConfig::GST PkgConfig::SRTP OpenSSL::Crypto)
add_executable(traverse-launcher launcher.cpp standalone.cpp)
target_link_libraries(traverse-launcher PRIVATE traverse-common traverse-settings traverse-tls-routing)
install(TARGETS traverse-adapter traverse-service traverse-runtime-check traverse-launcher traverse-tls-frontend traverse-bind-helper RUNTIME DESTINATION libexec/traverse)
install(DIRECTORY ../web/ DESTINATION share/traverse/web)

install(FILES ../packaging/config.example.json ../packaging/config.legacy.example.json DESTINATION share/traverse)
