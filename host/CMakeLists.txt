cmake_minimum_required(VERSION 3.20)
project(traverse_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/runtime-manifest.json" runtime_manifest)
string(JSON host_minimum_qt GET "${runtime_manifest}" hostUi minimumQt)
string(JSON host_minimum_openssl GET "${runtime_manifest}" hostUi minimumOpenSSL)
find_package(Qt6 ${host_minimum_qt} REQUIRED COMPONENTS Core Gui Widgets DBus)
find_package(nlohmann_json REQUIRED)
add_library(traverse-owner-client owner_client.cpp)
target_include_directories(traverse-owner-client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(traverse-owner-client PUBLIC Qt6::Core PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(traverse-owner-client PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(traverse-connection-feedback connection_feedback.cpp)
target_link_libraries(traverse-connection-feedback PUBLIC Qt6::Core)
target_compile_options(traverse-connection-feedback PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(traverse-host main.cpp sharing_window.cpp)
target_link_libraries(traverse-host PRIVATE traverse-owner-client traverse-connection-feedback Qt6::Widgets Qt6::DBus)
target_compile_options(traverse-host PRIVATE -Wall -Wextra -Wpedantic -Werror)
# An explicit package entry is separate from desktop/default-entry promotion.
install(TARGETS traverse-host RUNTIME DESTINATION libexec/traverse)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 ${host_minimum_qt} REQUIRED COMPONENTS Test)
  find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
  add_executable(sharing-feedback-test ../tests/sharing_feedback_test.cpp sharing_window.cpp)
  set_target_properties(sharing-feedback-test PROPERTIES AUTOMOC ON)
  target_link_libraries(sharing-feedback-test PRIVATE traverse-owner-client traverse-connection-feedback Qt6::Widgets Qt6::Test Qt6::DBus)
  target_compile_options(sharing-feedback-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME sharing-feedback COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:sharing-feedback-test>)
  set_tests_properties(sharing-feedback PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(host-window-driver ../tests/host_window_driver.cpp sharing_window.cpp)
  target_link_libraries(host-window-driver PRIVATE traverse-owner-client traverse-connection-feedback Qt6::Widgets Qt6::Test Qt6::DBus)
  target_compile_options(host-window-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(owner-client-test ../tests/owner_client_test.cpp)
  set_target_properties(owner-client-test PROPERTIES AUTOMOC ON)
  target_link_libraries(owner-client-test PRIVATE traverse-owner-client Qt6::Test)
  target_compile_options(owner-client-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME owner-client COMMAND owner-client-test)
  set_tests_properties(owner-client PROPERTIES TIMEOUT 30)
endif()
find_package(OpenSSL ${host_minimum_openssl} REQUIRED)
find_package(Threads REQUIRED)
add_library(traverse-ui-configuration configuration_store.cpp
  ../native/endpoint_settings.cpp ../native/certificate_settings.cpp ../native/observer_settings.cpp ../native/tls_certificate.cpp ../native/local_authority.cpp)
target_include_directories(traverse-ui-configuration PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../native)
target_link_libraries(traverse-ui-configuration PUBLIC nlohmann_json::nlohmann_json OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
target_compile_options(traverse-ui-configuration PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(traverse-ui-forms endpoint_form.cpp settings_dialog.cpp setup_window.cpp public_certificate.cpp)
target_link_libraries(traverse-ui-forms PUBLIC traverse-ui-configuration Qt6::Widgets)
target_compile_options(traverse-ui-forms PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(traverse-host PRIVATE traverse-ui-forms)
if(BUILD_TESTING)
  target_link_libraries(host-window-driver PRIVATE traverse-ui-forms)
  target_link_libraries(sharing-feedback-test PRIVATE traverse-ui-forms)
  add_executable(public-certificate-probe ../tests/public_certificate_probe.cpp)
  target_link_libraries(public-certificate-probe PRIVATE traverse-ui-forms)
  target_compile_options(public-certificate-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)

  set_target_properties(settings-dialog-test PROPERTIES AUTOMOC ON)
  target_link_libraries(settings-dialog-test PRIVATE traverse-ui-forms Qt6::Test)
  target_compile_options(settings-dialog-test PRIVATE -Wall -Wextra -Wpedantic -Werror)

  set_tests_properties(settings-dialog PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

  set_target_properties(setup-window-test PROPERTIES AUTOMOC ON)
  target_link_libraries(setup-window-test PRIVATE traverse-ui-forms Qt6::Test)

  set_tests_properties(setup-window PROPERTIES TIMEOUT 45 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
